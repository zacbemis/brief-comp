// each file is a module named after the file, with the extension .brf
// variables are default private and can be used in the same module
// modules are found by name, so no import is needed
// file: User.brf

pub User {
  name: str
  age: int
  userVer: flt
}

pub type Tier = Sym<:tier1, :tier2, :tier3>

pub trait HasUser{
  getUser(value: Self): User

  pub getName(value: Self): str {
    value.getUser().name
  }

  pub getAge(value: Self): int {
    value.getUser().age
  }

  pub getAgeRange(value: Self): Sym<:negative, :ofAge, :old> {
    value.getAge() |> {
      < 0 => :negative
      0..100 => :ofAge
      >= 100 => :old
    }
  }
}

impl User: HasUser {
  pub const VERSION: flt = 0.5

  getUser(user: User): User {user}

  pub new(name: str, age: int): Sym<:ok(User), :err(str)> {
    if age < 18 {
      ret :err("Age is too low")
    }

    :ok(User {
      name: name
      age: age
      userVer: VERSION 
    })
  }
  
  pub fromJson(json: Json): Sym<:ok(User), Json.Error, :err(str)> {
    let name = json.getStr("name")?
    let age = json.getInt("age")?

    User.new(name, age)? // a call that can fail is handled, even on the last line
  }

  pub toJson(user: User): Json {
    Json.object()
      |> set("name", Json.str(user.name))
      |> set("age", Json.int(user.age))
      |> set("Version", Json.flt(user.userVer))
  }
}

// file: Admin.brf

// User.brf is in the same folder, so User needs no import
// only the name User comes in; everything else is User.something

pub Admin {
  user: User
  userTier: User.Tier
}

impl Admin: User.HasUser {
  getUser(admin: Admin): User {admin.user}

  ovr pub getName(admin: Admin): str {
    admin.getUser().name
  }

  // ending with a plain value returns :ok of it
  pub new(name: str, age: int): Sym<:ok(Admin), :err(str)> {
    Admin { user: User.new(name, age)?, userTier: :tier1 }
  }
}

// file: UserCollection.brf

pub UserCollection {
  userVec: Vec<User>
  userVersion: flt
}

impl UserCollection {
  pub new(): UserCollection {
    UserCollection {
      userVec: Vec<User>()
      userVersion: User.VERSION
    }
  }

  pub add(collection: &UserCollection, user: User): () {
    collection.userVec.add(user)
  }

  pub fromJson(json: Json): Sym<:ok(UserCollection), Json.Error>{
    let users = json.getArray("users")?
    let collection = UserCollection.new()

    for item in users {
      let user = User.fromJson(item)?
      collection.add(user)
    }

    :ok(collection)
  }

  pub toJson(collection: UserCollection): Json {
    Json.object()
      |> set("users", Json.array(collection.userVec |> select((user) => user.toJson())))
  }

  pub findByName(collection: UserCollection, name: str): Sym<:found(User), :missing> {
    collection.userVec.find((user) => user.getName() == name) |> {
      :ok(user) => :found(user)
      :none => :missing
    }
  }

  // remove takes out every match and returns them
  pub removeByName(collection: &UserCollection, name: str): Vec<User> {
    collection.userVec |> remove((user) => user.getName() == name)
  }

  pub count(collection: UserCollection): int {
    collection.userVec.len()
  }

  pub olderThan(collection: UserCollection, age: int): Vec<User> {
    collection.userVec |> where((user) => user.getAge() > age)
  }
}
