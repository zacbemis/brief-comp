// variables are default private and can be used in the same module
mod User {
  import std.Json

  pub User {
    name: str
    age: int
    userVer: flt
  }

  pub trait HasUser{
    getUser(value: Self): User

    pub getName(value: Self): str {
      value.getUser().name
    }

    pub getAge(value: Self): int {
      value.getUser().age
    }

    pub getAgeRange(value: Self): Sym<:ofAge, :old> {
      value.getAge() |> {
        18..100 => :ofAge
        >= 100 => :old
      }
    }
  }

  impl User: HasUser {
    pub const VERSION = 0.5
    pub enum USER_TIER {
      TIER1
      TIER2
      TIER3
    }

    getUser(user: User): User {user}

    pub new(name: str, age: int): Sym<:ok(User), :err(str)> {
      if age < 18 {
        ret :err("Age is too low")
      }

      :ok(User {
        name: name
        age: age
        userVer: VERSION 
      })
    }
    
    pub fromJson(json: Json): Sym<:ok(User), Json.sym()> {
      let name = json.getStr("name")?
      let age = json.getInt("age")?

      User.new(name, age)
    }

    pub toJson(user: User): Json {
      Json.object({
        "name": user.name,
        "age": user.age,
        "Version: user.userVer,
      })
    }
  }
}

mod Admin {
  import User

  pub Admin {
    user: User
    userTier: USER_TIER
  }

  impl Admin: HasUser {
    getUser(admin: Admin): User {admin.user}

    ovr pub getName(admin: Admin): str {
      Admin.getUser().name
    }

    pub new(name: str, age: int): Admin {
      Admin {
        user: User.new(name, age)?
        userTier: USER_TIER.TIER1
      }
    }
  }
}

mod UserCollection {
  import std.Json

  import User

  pub UserCollection {
    userVec: Vec[User]
    userVersion: flt
  }

  impl UserCollection {
    pub new(): userCollection {
      UserCollection {
        userVec: Vec[User]()
        userVersion: User.VERSION
      }

      })
    }

    pub findByName(collection: UserCollection, name: str): Sym<:found(User), :missing> {
      collection.userVec.find((user) => user.getName() == name) |> {
        :ok(user) => :found(user)
        :none => :missing
      }
    }

    pub removeByName(collection: &UserCollection, name: str) Sym<:found(User), :missing> {
      collection.userVec.remove((user) => user.getName() == name) |> {
        :ok(user) => :found(user)
        :none => :missing
      }
    }

    pub count(collection: UserCollection): int {
      collection.userVec.len()
    }

    pub olderThan(collection: UserCollection, age: int): User[] {
      collection.userVec.filter((user) => user.getAge() > age)
    }
  }
}
