// each file is a module named after the file, with the extension .brf
// functions, constants and fields are private unless marked pub
// types, traits and symbols are always public
// modules are found by name, so no import is needed
// file: User.brf

User: HasUser {
  name: str
  pub age: int
  userVer: flt

  getUser() -> User { self }

  // does not use self: an associated function, called as User.new(...)
  pub new(name: str, age: int) -> :ok[User] | :!tooYoung[int] {
    if age < 18 {
      ret :!tooYoung[age]
    }

    User { name = name, age = age, userVer = version }
  }

  pub fromJson(json: Json) -> :ok[User] | Json.Error | :!tooYoung[int] {
    let name = json.getStr("name")?
    let age = json.getInt("age")?

    User.new(name, age) // an outcome on the last line is returned as it is
  }

  // uses self: a method, called as user.toJson()
  pub toJson() -> Json {
    Json.object()
      |> set("name", Json.str(self.name))
      |> set("age", Json.int(self.age))
      |> set("Version", Json.flt(self.userVer))
  }

  // changes self, so it takes &self
  pub birthday(&self) {
    self.age++
  }
}

pub const version: flt = 0.5

Tier = :tier1 | :tier2 | :tier3

// a declaration with functions and no fields is a trait
// a function without a body is required, one with a body is a default
HasUser {
  getUser(self) -> User

  getName(self) -> str {
    self.getUser().name
  }

  getAge(self) -> int {
    self.getUser().age
  }

  getAgeRange(self) -> :negative | :ofAge | :old {
    self.getAge() |> {
      < 0 => :negative
      0..100 => :ofAge
      >= 100 => :old
    }
  }
}

// file: Admin.brf

// User.brf is in the project, so User needs no import
// only the name User comes in; everything else is User.something

Admin: User.HasUser {
  user: User
  userTier: User.Tier

  getUser() -> User { self.user }

  // replaces the default getName from HasUser
  getName() -> str {
    "Admin " + self.user.getName()
  }

  // ending with a plain value returns :ok of it
  pub new(name: str, age: int) -> :ok[Admin] | User.:!tooYoung[int] {
    Admin { user = User.new(name, age)?, userTier = :tier1 }
  }
}

// file: UserCollection.brf

UserCollection {
  userVec: Vec[User]
  userVersion: flt

  pub new() -> UserCollection {
    UserCollection {
      userVec = Vec[User]()
      userVersion = User.version
    }
  }

  pub add(&self, user: &User) -> () {
    self.userVec.add(user)
  }

  pub fromJson(json: Json) -> :ok[UserCollection] | Json.Error | User.:!tooYoung[int] {
    let users = json.getArray("users")?
    let collection = UserCollection.new()

    for item in users {
      let user = User.fromJson(item)?
      collection.add(user)
    }

    collection
  }

  pub toJson() -> Json {
    Json.object()
      |> set("users", Json.array(self.userVec |> select((user) => user.toJson())))
  }

  pub findByName(name: str) -> :found[User] | :missing {
    self.userVec.find((user) => user.getName() == name) |> {
      :ok[user] => :found[user]
      :none => :missing
    }
  }

  // remove takes out every match and returns them
  pub removeByName(&self, name: str) -> Vec[User] {
    self.userVec |> remove((user) => user.getName() == name)
  }

  pub count() -> int {
    self.userVec.len()
  }

  pub olderThan(age: int) -> Vec[User] {
    self.userVec |> where((user) => user.getAge() > age)
  }
}
